\section{Input Normalization}
% =============================================================================

Each OD pair is bounded in $[0, 2000]$. Before GP training every point is mapped
to the unit hypercube $[0,1]^d$.

\begin{algorithm}[H]
\caption{Input normalization / unnormalization}
\begin{algorithmic}[1]
\Function{Normalize}{$\mathbf{X}_{\mathrm{real}},\, \mathbf{b}_{\mathrm{lo}},\, \mathbf{b}_{\mathrm{hi}}$}
    \State \Return $\dfrac{\mathbf{X}_{\mathrm{real}} - \mathbf{b}_{\mathrm{lo}}}{\mathbf{b}_{\mathrm{hi}} - \mathbf{b}_{\mathrm{lo}}}$
    \Comment{maps $[0,2000]^d \to [0,1]^d$}
\EndFunction
\vspace{4pt}
\Function{Unnormalize}{$\mathbf{X}_{\mathrm{norm}},\, \mathbf{b}_{\mathrm{lo}},\, \mathbf{b}_{\mathrm{hi}}$}
    \State \Return $\mathbf{X}_{\mathrm{norm}} \cdot (\mathbf{b}_{\mathrm{hi}} - \mathbf{b}_{\mathrm{lo}}) + \mathbf{b}_{\mathrm{lo}}$
\EndFunction
\end{algorithmic}
\end{algorithm}

\noindent\textbf{Initial design.} $N_{\mathrm{init}}$ points are drawn with a scrambled
\textbf{Sobol} quasi-random sequence in $[0,1]^d$.

% =============================================================================
\section{Output Standardization (per edge, external)}
% =============================================================================

Standardization is performed \emph{outside} the GP model so the graph kernel and
ICM model both receive zero-mean, unit-variance targets per task.

\begin{algorithm}[H]
\caption{Per-edge output standardization}
\begin{algorithmic}[1]
\Require $\mathbf{Y}_{\mathrm{raw}} \in \mathbb{R}^{N \times L}$ \quad (may contain \texttt{NaN})
\For{$\ell = 1, \dots, L$}
    \State $\mu_\ell \gets \operatorname{mean}\bigl(\mathbf{Y}_{\mathrm{raw}}[\cdot,\ell],\; \text{ignore NaN}\bigr)$
    \State $\sigma_\ell \gets \max\!\Bigl(\operatorname{std}\bigl(\mathbf{Y}_{\mathrm{raw}}[\cdot,\ell],\; \text{ignore NaN}\bigr),\; 10^{-6}\Bigr)$
    \State $\mathbf{Y}_{\mathrm{std}}[\cdot,\ell] \gets \dfrac{\mathbf{Y}_{\mathrm{raw}}[\cdot,\ell] - \mu_\ell}{\sigma_\ell}$
\EndFor
\State \Return $\mathbf{Y}_{\mathrm{std}},\; \bm{\mu} \in \mathbb{R}^L,\; \bm{\sigma} \in \mathbb{R}^L$
\end{algorithmic}
\end{algorithm}

\section{Graph Construction}
% =============================================================================

\subsection{Adjacency matrix}

\begin{algorithm}[H]
\caption{Adjacency from SUMO connections (driving connectivity)}
\begin{algorithmic}[1]
\Require \texttt{net.xml}, edge list $\{e_1,\dots,e_L\}$
\State $\mathbf{A} \gets \mathbf{0}_{L\times L}$
\ForEach{$\langle\texttt{connection from}=f\;\texttt{to}=t\rangle$ in \texttt{net.xml}}
    \If{$f \in \{e_\ell\}$ \textbf{and} $t \in \{e_\ell\}$ \textbf{and} $f \neq t$}
        \State $\mathbf{A}[\mathrm{idx}(f),\, \mathrm{idx}(t)] \gets 1$
        \State $\mathbf{A}[\mathrm{idx}(t),\, \mathrm{idx}(f)] \gets 1$ \Comment{undirected}
    \EndIf
\EndFor
\State $\mathbf{A} \gets \mathbf{A} - \operatorname{diag}(\mathbf{A})$ \Comment{zero diagonal}
\State \Return $\mathbf{A}$
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{Adjacency by shared junction node (weighted)}
\begin{algorithmic}[1]
\Require \texttt{net.xml}, edge list $\{e_\ell\}$, \texttt{normalize} $\in$ \{``k-1'', ``k''\}
\State $\mathbf{A} \gets \mathbf{0}_{L\times L}$; \quad $\mathcal{N} \gets \{\}$
\ForEach{$\langle\texttt{edge id}=e\;\texttt{from}=u\;\texttt{to}=v\rangle$}
    \State $\mathcal{N}[u] \mathrel{+}= \{\mathrm{idx}(e)\}$;\quad $\mathcal{N}[v] \mathrel{+}= \{\mathrm{idx}(e)\}$
\EndFor
\ForEach{node $n$ with incident edge set $\mathcal{E}_n$, $|\mathcal{E}_n| = k > 1$}
    \State $w \gets 1\,/\,(k-1)$ \Comment{weight per edge pair}
    \For{all pairs $(i,j) \in \mathcal{E}_n \times \mathcal{E}_n,\; i \neq j$}
        \State $\mathbf{A}[i,j] \mathrel{+}= w$
    \EndFor
\EndFor
\State $\mathbf{A} \gets \min(\mathbf{A},\; 1)$; \quad $\mathbf{A} \gets \mathbf{A} - \operatorname{diag}(\mathbf{A})$
\State \Return $\mathbf{A} \in [0,1]^{L\times L}$
\end{algorithmic}
\end{algorithm}

\subsection{Laplacian}

\begin{algorithm}[H]
\caption{Graph Laplacian (normalized or unnormalized)}
\begin{algorithmic}[1]
\Require $\mathbf{A}$, \texttt{normalized} $\in \{\texttt{True}, \texttt{False}\}$
\State $\mathbf{A} \gets \tfrac{1}{2}(\mathbf{A} + \mathbf{A}^\top)$ \Comment{symmetrize}
\State $\mathbf{d} \gets \mathbf{A}\,\mathbf{1}$ \Comment{degree vector}
\If{\texttt{normalized}}
    \State $\tilde{\mathbf{d}} \gets \mathbf{d}^{-1/2}$;\quad
           $\tilde{\mathbf{A}} \gets \operatorname{diag}(\tilde{\mathbf{d}})\,\mathbf{A}\,\operatorname{diag}(\tilde{\mathbf{d}})$
    \State $\mathbf{L} \gets \mathbf{I} - \tilde{\mathbf{A}}$ \Comment{$\lambda_i \in [0,2]$}
\Else
    \State $\mathbf{L} \gets \operatorname{diag}(\mathbf{d}) - \mathbf{A}$
\EndIf
\State \Return $\tfrac{1}{2}(\mathbf{L}+\mathbf{L}^\top)$ \Comment{enforce numerical symmetry}
\end{algorithmic}
\end{algorithm}

\subsection{Shared eigendecomposition}

\begin{algorithm}[H]
\caption{Laplacian eigendecomposition (used by all graph kernels)}
\begin{algorithmic}[1]
\Require $\mathbf{A}$, \texttt{spectrum\_k} (optional low-pass truncation)
\State $(\bm{\lambda}, \mathbf{V}) \gets \operatorname{eigh}\!\bigl(\mathbf{L}(\mathbf{A})\bigr)$ \Comment{ascending order}
\State $\bm{\lambda} \gets \max(\bm{\lambda},\, 0)$
\If{\texttt{spectrum\_k} is set}
    \State Keep only the $k$ smallest pairs: $\bm{\lambda} \gets \bm{\lambda}_{:k}$,\;
           $\mathbf{V} \gets \mathbf{V}_{:,:k}$ \Comment{low-pass filter}
\EndIf
\State \Return $(\bm{\lambda} \in \mathbb{R}^k,\; \mathbf{V} \in \mathbb{R}^{L\times k})$
\end{algorithmic}
\end{algorithm}

\section{Graph Task Kernels $\mathbf{K}_G \in \mathbb{R}^{L\times L}$}
% =============================================================================

\subsection{Diffusion kernel (heat kernel)}

\[
\mathbf{K}_G = \mathbf{V}\,\operatorname{diag}\!\bigl(e^{-\tau\bm{\lambda}}\bigr)\,\mathbf{V}^\top,
\qquad \tau > 0.
\]

\begin{algorithm}[H]
\caption{Diffusion kernel}
\begin{algorithmic}[1]
\Require $\mathbf{A}$, \texttt{tau\_mode} $\in$ \{``auto'', ``fixed''\}, \texttt{tau\_scale}
\State $(\bm{\lambda}, \mathbf{V}) \gets \textsc{LaplacianEigh}(\mathbf{A})$
\If{\texttt{tau\_mode} = ``auto''}
    \State $s \gets \operatorname{median}\{\lambda_i : \lambda_i > 10^{-12}\}$
           \Comment{neutral scale if no positive eigenvalue}
    \State $\tau \gets \texttt{tau\_scale} / s$
\Else
    \State $\tau \gets \tau_{\mathrm{fixed}}$
\EndIf
\State $\mathbf{K}_G \gets \mathbf{V}\,\operatorname{diag}(e^{-\tau\bm{\lambda}})\,\mathbf{V}^\top$
\State $\mathbf{K}_G \gets \tfrac{1}{2}(\mathbf{K}_G + \mathbf{K}_G^\top)$
\State \Return $(\mathbf{K}_G,\; \tau)$
\end{algorithmic}
\end{algorithm}

\noindent\textbf{Learnable variant.} $\tau$ is stored as a trainable parameter via the
softplus reparameterization $\tau = \operatorname{softplus}(\tilde{\tau})$ and updated
during MLL optimisation.

\subsection{Regularized Laplacian kernel}

\[
\mathbf{K}_G = (\mathbf{I} + \gamma\,\mathbf{L})^{-1}
= \mathbf{V}\,\operatorname{diag}\!\left(\frac{1}{1+\gamma\bm{\lambda}}\right)\mathbf{V}^\top,
\qquad \gamma > 0.
\]

\begin{algorithm}[H]
\caption{Regularized Laplacian kernel}
\begin{algorithmic}[1]
\Require $\mathbf{A}$, \texttt{gamma\_mode}, \texttt{gamma\_scale}
\State $(\bm{\lambda}, \mathbf{V}) \gets \textsc{LaplacianEigh}(\mathbf{A})$
\If{\texttt{gamma\_mode} = ``auto''}
    \State $s \gets \operatorname{median}\{\lambda_i > 10^{-12}\}$;\quad
           $\gamma \gets \texttt{gamma\_scale} / s$
\EndIf
\State $\mathbf{K}_G \gets \mathbf{V}\,\operatorname{diag}\!\left(\tfrac{1}{1+\gamma\bm{\lambda}}\right)\mathbf{V}^\top$
\State $\mathbf{K}_G \gets \tfrac{1}{2}(\mathbf{K}_G + \mathbf{K}_G^\top)$
\State \Return $(\mathbf{K}_G,\; \gamma)$
\end{algorithmic}
\end{algorithm}

\subsection{Random-walk kernel}

\[
\mathbf{K}_G = \mathbf{V}\,\operatorname{diag}\!\bigl((a - \bm{\lambda})^p\bigr)\,\mathbf{V}^\top,
\qquad a \geq \lambda_{\max},\; p \in \mathbb{Z}_{>0}.
\]

\begin{algorithm}[H]
\caption{Random-walk kernel (fixed)}
\begin{algorithmic}[1]
\Require $\mathbf{A}$, $p \geq 1$, $a \geq \lambda_{\max}$
\State $(\bm{\lambda}, \mathbf{V}) \gets \textsc{LaplacianEigh}(\mathbf{A})$
\State \textbf{assert} $a \geq \max(\bm{\lambda})$ \Comment{ensures PSD}
\State $\mathbf{K}_G \gets \mathbf{V}\,\operatorname{diag}\!\bigl((a-\bm{\lambda})^p\bigr)\,\mathbf{V}^\top$
\State $\mathbf{K}_G \gets \tfrac{1}{2}(\mathbf{K}_G + \mathbf{K}_G^\top)$
\State $\mathbf{K}_G \gets \mathbf{K}_G \,/\, \operatorname{mean}(\operatorname{diag}(\mathbf{K}_G))$
\Comment{normalize scale}
\State \Return $\mathbf{K}_G$
\end{algorithmic}
\end{algorithm}

\section{GP Models}
% =============================================================================

\subsection{Graph-kernel GP (product kernel)}

Training observations are \emph{stacked}: each row is $(\mathbf{x}, \ell)$ where $\ell$
is an integer task index appended as the last feature column.

\[
k\!\bigl((\mathbf{x},\ell),(\mathbf{x}',\ell')\bigr)
= k_X(\mathbf{x},\mathbf{x}') \cdot K_G(\ell,\ell')
\]

\begin{algorithm}[H]
\caption{\textsc{BuildGraphMultitaskGP}}
\begin{algorithmic}[1]
\Require \texttt{net.xml}, $\{e_\ell\}$, $\tilde{\mathbf{X}} \in \mathbb{R}^{NL \times (d+1)}$,
         $\tilde{\mathbf{Y}} \in \mathbb{R}^{NL \times 1}$
\State $\mathbf{A} \gets$ \textsc{Adjacency}(\texttt{net.xml}, $\{e_\ell\}$)
\State $\mathbf{K}_G \gets$ \textsc{GraphKernel}($\mathbf{A}$) \Comment{diffusion / reg.\ Lap.\ / RW}

\vspace{4pt}
\State \textit{// Per-task standardization of $\tilde{\mathbf{Y}}$}
\For{$\ell = 0,\dots,L-1$}
    \State $\text{mask}_\ell \gets \bigl\{i : \tilde{X}_{i,d} = \ell,\; \tilde{Y}_i \neq \texttt{NaN}\bigr\}$
    \State $\mu_\ell \gets \operatorname{mean}(\tilde{Y}[\text{mask}_\ell])$;\quad
           $\sigma_\ell \gets \operatorname{std}(\tilde{Y}[\text{mask}_\ell]).\operatorname{clamp}(10^{-6})$
    \State $\tilde{Y}[\text{mask}_\ell] \gets (\tilde{Y}[\text{mask}_\ell] - \mu_\ell)\,/\,\sigma_\ell$
\EndFor

\vspace{4pt}
\State \textit{// Product kernel with split active dims}
\State $k_X \gets \textsc{ScaleKernel}\!\bigl(\textsc{MaternKernel}(\nu{=}2.5,\,\text{ARD})\bigr)$;\quad
       $k_X.\texttt{active\_dims} \gets [0,\dots,d{-}1]$
\State $k_T \gets \textsc{FixedGraphTaskKernel}(\mathbf{K}_G)$;\quad
       $k_T.\texttt{active\_dims} \gets [d]$
\State $k \gets k_X \cdot k_T$

\vspace{4pt}
\State $\mathcal{M} \gets \textsc{SingleTaskGP}(\tilde{\mathbf{X}},\, \tilde{\mathbf{Y}},\, \text{covar}=k,\,
       \text{outcome\_transform}=\texttt{None})$
\State $\mathcal{L} \gets \textsc{ExactMarginalLogLikelihood}(\mathcal{M})$
\State \textsc{Fit}($\mathcal{L}$) \Comment{Algorithm~\ref{alg:fit}}
\State \Return $\mathcal{M}$
\end{algorithmic}
\end{algorithm}

\subsection{ICM multi-output GP}

\[
k_{\mathrm{ICM}}\!\bigl(\mathbf{x},\ell,\mathbf{x}',\ell'\bigr)
= k_X(\mathbf{x},\mathbf{x}') \cdot \bigl(\mathbf{B}\mathbf{B}^\top + \operatorname{diag}(\mathbf{v})\bigr)_{\ell\ell'}
\]
where $\mathbf{B} \in \mathbb{R}^{L\times r}$ and $\mathbf{v} \in \mathbb{R}^L$ are
learnable task-covariance parameters (ICM factorization, rank $r$).

\begin{algorithm}[H]
\caption{\textsc{InitializeICMGP}}
\begin{algorithmic}[1]
\Require $\mathbf{X}_{\mathrm{norm}} \in \mathbb{R}^{N\times d}$,\;
         $\mathbf{Y}_{\mathrm{std}} \in \mathbb{R}^{N\times L}$,\; rank $r$
\State $\text{likelihood} \gets \textsc{MultitaskGaussianLikelihood}(L,\; \sigma^2 \in [10^{-4},1])$
\State $k_{\mathrm{data}} \gets \textsc{MaternKernel}(\nu{=}2.5,\,\text{ARD},\, \ell \in [0.005,4])$
\State $k_{\mathrm{ICM}} \gets \textsc{MultitaskKernel}(k_{\mathrm{data}},\, L,\, \text{rank}=r)$
\State $\mathcal{M} \gets \textsc{ICMMultiOutputGP}(\mathbf{X}_{\mathrm{norm}},\,\mathbf{Y}_{\mathrm{std}},\,
       k_{\mathrm{ICM}},\, \text{likelihood})$
\State $\mathcal{L} \gets \textsc{ExactMarginalLogLikelihood}(\mathcal{M})$
\State \Return $(\mathcal{M},\, \mathcal{L})$
\end{algorithmic}
\end{algorithm}

\section{GP Fitting}
% =============================================================================

\begin{algorithm}[H]
\caption{\textsc{Fit} — L-BFGS-B with Adam fallback}
\label{alg:fit}
\begin{algorithmic}[1]
\Require marginal log-likelihood $\mathcal{L}$
\Try
    \State \textsc{FitGPyTorchMLL}($\mathcal{L}$) \Comment{L-BFGS-B, BoTorch default}
\Catch{any exception}
    \State $\mathcal{L}.\textsc{train}()$
    \State $\theta \gets \text{Adam}(\mathcal{L}.\text{parameters}(),\; lr=0.1)$
    \For{$t = 1,\dots,200$}
        \State $\theta.\textsc{zero\_grad}()$
        \State $\hat{\mathbf{y}} \gets \mathcal{M}(\mathbf{X}_{\mathrm{train}})$
        \State $\mathcal{J} \gets -\mathcal{L}(\hat{\mathbf{y}},\,\mathbf{Y}_{\mathrm{train}}).\operatorname{sum}()$
        \State $\mathcal{J}.\textsc{backward}()$;\quad $\theta.\textsc{step}()$
    \EndFor
\EndTry
\end{algorithmic}
\end{algorithm}

\section{Scalar Aggregation for Acquisition}
% =============================================================================

Both graph-GP and ICM models are wrapped to expose a \emph{scalar} surrogate
$S(\mathbf{x}) = \sum_\ell w_\ell\, z_\ell(\mathbf{x})$ compatible with
scalar acquisition functions.

\begin{algorithm}[H]
\caption{\textsc{LinearAggregation.posterior}($\mathbf{X}$)}
\begin{algorithmic}[1]
\Require $\mathbf{X} \in \mathbb{R}^{b\times d}$, weights $\mathbf{w} \in \mathbb{R}^L$
\For{$i = 1, \dots, b$}
    \State $\mathcal{N}(\bm{\mu}_i,\,\bm{\Sigma}_i) \gets \mathcal{M}(\mathbf{x}_i)$
    \Comment{multitask posterior at $\mathbf{x}_i$; $\bm{\mu}_i \in \mathbb{R}^L$, $\bm{\Sigma}_i \in \mathbb{R}^{L\times L}$}
    \State $\mu_{S,i} \gets \mathbf{w}^\top \bm{\mu}_i$
    \State $\sigma^2_{S,i} \gets \mathbf{w}^\top \bm{\Sigma}_i\, \mathbf{w}$;\quad
           $\sigma^2_{S,i} \gets \max(\sigma^2_{S,i},\, 0)$
\EndFor
\State \Return $\mathcal{N}\!\bigl(\bm{\mu}_S,\, \operatorname{diag}(\bm{\sigma}^2_S)\bigr)$
\Comment{scalar posterior $\in \mathbb{R}^{b\times 1}$}
\end{algorithmic}
\end{algorithm}

\noindent\textbf{Effective weights by model:}
\begin{itemize}
\item \textbf{Uniform}: $w_\ell = 1/L$
\item \textbf{Flow-proportional}: $w_\ell = f_\ell \,/\, (L \sum_{\ell'} f_{\ell'})$, where $f_\ell$ is the ground-truth flow on edge $\ell$
\item \textbf{ICM / Graph-GP} (standardized space): effective weight $\tilde{w}_\ell = w_\ell \sigma_\ell$, so $S$ lives in Z-space and $f^* = \min_n \sum_\ell \tilde{w}_\ell z_\ell(\mathbf{x}_n)$
\end{itemize}

\section{Acquisition Function: Expected Improvement}
% =============================================================================

\begin{algorithm}[H]
\caption{\textsc{OptimizeEI}}
\begin{algorithmic}[1]
\Require aggregated model $\mathcal{M}_S$, incumbent $f^*$
\State $\alpha \gets \textsc{ExpectedImprovement}(\mathcal{M}_S,\; f^*,\; \text{maximize}=\texttt{False})$
\State $\mathbf{x}^* \gets \operatorname*{arg\,max}_{\mathbf{x} \in [0,1]^d}\ \alpha(\mathbf{x})$
\Comment{\textsc{optimize\_acqf}: 5 restarts, 32 raw samples}
\State \Return $\mathbf{x}^*$
\end{algorithmic}
\end{algorithm}

\section{Full BO Loop}
% =============================================================================

\begin{algorithm}[H]
\caption{\textsc{BayesianOptimizationLoop}}
\begin{algorithmic}[1]
\Require initial data $\{(\mathbf{x}^{(n)}_{\mathrm{norm}},\, \mathbf{e}^{(n)})\}_{n=1}^{N_0}$,
         $T$ iterations, patience $P$, threshold $\delta$

\State $\mathcal{D} \gets \{(\mathbf{x}^{(n)}_{\mathrm{norm}},\, \mathbf{e}^{(n)})\}$;\quad
       $S^* \gets \min_n \sum_\ell w_\ell e^{(n)}_\ell$

\For{$t = 1, \dots, T$}

    \State \textbf{--- Step 1: Standardize outputs ---}
    \State $(\mathbf{Y}_{\mathrm{std}},\bm{\mu},\bm{\sigma}) \gets
           \textsc{StandardizePerEdge}(\mathbf{E})$

    \State \textbf{--- Step 2: Fit GP ---}
    \If{model = \textit{Graph-GP}}
        \State $\tilde{\mathbf{X}} \gets \textsc{StackWithTaskIds}(\mathbf{X},\, L)$
        \Comment{$[NL, d+1]$}
        \State $\mathcal{M} \gets \textsc{BuildGraphMultitaskGP}(\tilde{\mathbf{X}},\,\mathbf{Y}_{\mathrm{std}})$
        \State $\mathbf{w}_{\mathrm{eff}} \gets \mathbf{w} \odot \bm{\sigma}$;\quad
               $f^* \gets \min_n \mathbf{w}_{\mathrm{eff}}^\top \mathbf{z}^{(n)}$
    \ElsIf{model = \textit{ICM}}
        \State $\mathcal{M} \gets \textsc{InitializeICMGP}(\mathbf{X},\,\mathbf{Y}_{\mathrm{std}},\,r{=}1)$
        \State \textsc{Fit}($\mathcal{L}$) \Comment{Algorithm~\ref{alg:fit}}
        \State $\mathbf{w}_{\mathrm{eff}} \gets \mathbf{w} \odot \bm{\sigma}$;\quad
               $f^* \gets \min_n \mathbf{w}_{\mathrm{eff}}^\top \mathbf{z}^{(n)}$
    \ElsIf{model = \textit{IndependentGP}}
        \For{$\ell = 1, \dots, L$}
            \State $\mathcal{M}_\ell \gets \textsc{SingleTaskGP}(\mathbf{X},\, \mathbf{e}_\ell)$
        \EndFor
        \State $f^* \gets \min_n \sum_\ell w_\ell e^{(n)}_\ell$
    \ElsIf{model = \textit{VanillaBO}}
        \State $\mathbf{s} \gets \mathbf{E}\,\mathbf{w}$;\quad
               $\mathcal{M} \gets \textsc{SingleTaskGP}(\mathbf{X},\,\mathbf{s})$
        \State $f^* \gets \min(\mathbf{s})$
    \EndIf

    \vspace{4pt}
    \State \textbf{--- Step 3: Optimize acquisition ---}
    \State $\mathcal{M}_S \gets \textsc{LinearAggregation}(\mathcal{M},\,\mathbf{w}_{\mathrm{eff}})$
    \State $\mathbf{x}^*_{\mathrm{norm}} \gets \textsc{OptimizeEI}(\mathcal{M}_S,\, f^*)$

    \vspace{4pt}
    \State \textbf{--- Step 4: Simulate \& evaluate ---}
    \State $\mathbf{x}^*_{\mathrm{real}} \gets \textsc{Unnormalize}(\mathbf{x}^*_{\mathrm{norm}})$
    \State \textsc{RunSUMO}($\mathbf{x}^*_{\mathrm{real}}$)
    \State $\mathbf{e}^* \gets \textsc{ComputePerEdgeError}(\text{sim},\, \text{GT})$
    \State $S^*_{\mathrm{new}} \gets \sum_\ell w_\ell e^*_\ell$

    \vspace{4pt}
    \State \textbf{--- Step 5: Update dataset ---}
    \State $\mathcal{D} \mathrel{+}= \{(\mathbf{x}^*_{\mathrm{norm}},\, \mathbf{e}^*)\}$;\quad
           $S^* \gets \min(S^*,\, S^*_{\mathrm{new}})$;\quad
           $\texttt{curve}[t] \gets S^*$

    \vspace{4pt}
    \State \textbf{--- Step 6: Early stopping ---}
    \If{no improvement $\geq \delta$ for last $P$ iterations}
        \State \textbf{break}
    \EndIf
\EndFor
\State \Return \texttt{BOResults}$(\mathcal{D},\, S^*,\, \texttt{curve}, \dots)$
\end{algorithmic}
\end{algorithm}

\section{Multiple Restarts}
% =============================================================================

\begin{algorithm}[H]
\caption{\textsc{RunMultipleRestarts}}
\begin{algorithmic}[1]
\Require $R$ restarts, $T$ BO iterations
\For{$r = 0, \dots, R-1$}
    \State $\text{seed}_r \gets 42 + r \times 1000$;\quad \textsc{SetSeed}($\text{seed}_r$)
    \State $\texttt{results}_r \gets \textsc{BayesianOptimizationLoop}(\dots,\, T)$
    \State \textsc{Save}($\texttt{results}_r,\;$
           \texttt{restart\_}$(r{+}1)$\texttt{\_seed-}$\text{seed}_r$)
\EndFor
\State \Return $\bigl\{\texttt{results}_r\bigr\}_{r=0}^{R-1}$,\quad
       $\bar{S}(t) = \tfrac{1}{R}\sum_r \texttt{curve}_r(t)$,\quad
       $r^* = \operatorname{arg\,min}_r S^*_r$
\end{algorithmic}
\end{algorithm}
